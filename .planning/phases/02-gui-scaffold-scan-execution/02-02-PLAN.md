   """Shared header bar — logo + app title + plan badge."""
   import customtkinter as ctk
   from pathlib import Path
   from app.config import DEV_MODE, APP_VERSION
   from app.gui.theme import (
       BG_PRIMARY, ACCENT_RED, TEXT_PRIMARY, TEXT_SECONDARY,
       font_heading, font_body,
   )

   LOGO_PATH = Path(__file__).resolve().parents[3] / "rakoon_logo.png"

   class HeaderBar(ctk.CTkFrame):
       def __init__(self, master, **kwargs):
           super().__init__(master, fg_color=BG_PRIMARY, height=56, **kwargs)
           self.pack_propagate(False)
           self._build()

       def _build(self) -> None:
           # Logo
           self._logo_image = None
           try:
               from PIL import Image
               if LOGO_PATH.exists():
                   logo_pil = Image.open(str(LOGO_PATH)).resize((36, 36))
                   self._logo_image = ctk.CTkImage(
                       light_image=logo_pil, dark_image=logo_pil, size=(36, 36)
                   )
           except ImportError:
               pass  # Pillow not installed — skip logo

           if self._logo_image:
               logo_label = ctk.CTkLabel(self, image=self._logo_image, text="")
               logo_label.pack(side="left", padx=(16, 8), pady=10)

           # Title
           title = ctk.CTkLabel(
               self, text="SecScan", font=font_heading(20),
               text_color=TEXT_PRIMARY,
           )
           title.pack(side="left", padx=(4, 0), pady=10)

           # Version
           ver = ctk.CTkLabel(
               self, text=f"v{APP_VERSION}", font=font_body(10),
               text_color=TEXT_SECONDARY,
           )
           ver.pack(side="left", padx=(6, 0), pady=(16, 10))

           # DEV badge
           if DEV_MODE:
               badge = ctk.CTkLabel(
                   self, text="DEV",
                   fg_color=ACCENT_RED, text_color="#ffffff",
                   corner_radius=4,
                   font=ctk.CTkFont(family="Poppins", size=10, weight="bold"),
                   width=36, height=20,
               )
               badge.pack(side="right", padx=16, pady=18)

1. Create `app/gui/app.py` — the main CTk application window with frame-switching navigation:

   ```python
   """SecScan main application window — single-window frame switching."""
   import customtkinter as ctk
   from app.gui.theme import BG_PRIMARY
   from app.gui.widgets.header_bar import HeaderBar

   ctk.set_appearance_mode("dark")

   class SecScanApp:
       """Root application window. Screens register as CTkFrame subclasses."""

       def __init__(self) -> None:
           self.root = ctk.CTk()
           self.root.title("SecScan")
           self.root.geometry("900x640")
           self.root.configure(fg_color=BG_PRIMARY)
           self.root.minsize(800, 560)

           # Header bar (fixed at top, shared across all screens)
           self._header = HeaderBar(self.root)
           self._header.pack(fill="x", side="top")

           # Content area for screen frames
           self._content = ctk.CTkFrame(self.root, fg_color=BG_PRIMARY)
           self._content.pack(fill="both", expand=True)

           self._screens: dict[str, ctk.CTkFrame] = {}
           self._active_screen: ctk.CTkFrame | None = None

       def register_screen(self, name: str, screen: ctk.CTkFrame) -> None:
           """Register a screen frame. Screen must already be created with _content as master."""
           self._screens[name] = screen

       def show_screen(self, name: str, **kwargs) -> None:
           """Switch to named screen. Calls on_show(kwargs) if screen has it."""
           if self._active_screen:
               self._active_screen.pack_forget()
           screen = self._screens[name]
           screen.pack(in_=self._content, fill="both", expand=True)
           self._active_screen = screen
           if hasattr(screen, "on_show"):
               screen.on_show(**kwargs)

       @property
       def content_frame(self) -> ctk.CTkFrame:
           """Parent frame for all screens."""
           return self._content

       def run(self) -> None:
           """Start the application main loop."""
           self.show_screen("home")
           self.root.mainloop()
   ```

   Key design decisions:
   - `set_appearance_mode("dark")` called at module level before CTk() instantiation.
   - HeaderBar is persistent at top, never hidden during screen switches.
   - `_content` frame is the parent for all screen frames. Screens pack INTO _content.
   - `show_screen` supports optional `on_show(**kwargs)` callback on screens for passing data (e.g., ScanResult to results screen).
   - Does NOT import any screens yet — screens register themselves in app/main.py wiring (Plan 02-05).
