```python
def on_show(self, scan_result: Optional[ScanResult] = None, **kwargs) -> None:
    self._result = scan_result
    # Clear old cards
    if self._cards_frame:
        for widget in self._cards_frame.winfo_children():
            widget.destroy()

    if not scan_result:
        self._total_label.configure(text="Sem resultado")
        return

    total = len(scan_result.findings)
    self._total_label.configure(text=f"{total} findings")

    # Per-scanner breakdown
    counts = Counter(f.tool for f in scan_result.findings)
    for tool in scan_result.tools_used:
        count = counts.get(tool, 0)
        self._create_tool_card(tool, count)

    # Scan info
    duration = f"{scan_result.scan_duration_seconds:.1f}s"
    files = scan_result.scanned_files
    self._info_label.configure(
        text=f"{files} arquivos analisados em {duration}"
    )
```

**_create_tool_card method** `(self, tool_name: str, count: int)`:
```python
def _create_tool_card(self, tool_name: str, count: int) -> None:
    card = ctk.CTkFrame(
        self._cards_frame, fg_color=BG_CARD,
        corner_radius=8, border_width=1, border_color=BG_BORDER,
        width=180, height=90,
    )
    card.pack_propagate(False)
    card.pack(side="left", padx=8, pady=8)

    ctk.CTkLabel(
        card, text=tool_name.capitalize(),
        font=font_body(13), text_color=TEXT_SECONDARY,
    ).pack(pady=(12, 2))

    color = ACCENT_RED if count > 0 else ACCENT_TEAL
    ctk.CTkLabel(
        card, text=str(count),
        font=font_heading(24), text_color=color,
    ).pack(pady=(2, 4))

    ctk.CTkLabel(
        card, text="findings",
        font=font_body(10), text_color=TEXT_SECONDARY,
    ).pack()
```

   ```python
   """SecScan — Entry point."""
   from app.gui.app import SecScanApp
   from app.gui.screens.main_screen import HomeScreen
   from app.gui.screens.settings_screen import SettingsScreen
   from app.gui.screens.progress_screen import ProgressScreen
   from app.gui.screens.results_screen import ResultsScreen


   def main() -> None:
       app = SecScanApp()

       # Navigation callback — screens call this to switch
       def navigate(screen_name: str, **kwargs) -> None:
           app.show_screen(screen_name, **kwargs)

       # Create and register screens
       home = HomeScreen(app.content_frame, nav_callback=navigate)
       app.register_screen("home", home)

       settings = SettingsScreen(app.content_frame, nav_callback=navigate)
       app.register_screen("settings", settings)

       progress = ProgressScreen(app.content_frame, nav_callback=navigate)
       app.register_screen("progress", progress)

       results = ResultsScreen(app.content_frame, nav_callback=navigate)
       app.register_screen("results", results)

       app.run()


   if __name__ == "__main__":
       main()
   ```
