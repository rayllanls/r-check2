```python
def _start_scan(self) -> None:
    path = self._path_entry.get().strip()
    url = self._url_entry.get().strip()
    target = path or url
    if not target:
        from tkinter import messagebox
        messagebox.showwarning("SecScan", "Informe um caminho ou URL.")
        return
    self._navigate("progress", target=target, plan=self._plan)
```

**_load_token method**:
```python
import json
from app.config import CONFIG_FILE

def _load_token(self) -> None:
    try:
        if CONFIG_FILE.exists():
            data = json.loads(CONFIG_FILE.read_text())
            token = data.get("groq_token", "")
            self._token_entry.delete(0, "end")
            if token:
                self._token_entry.insert(0, token)
    except (json.JSONDecodeError, OSError):
        pass
```

**_save_token method**:
```python
def _save_token(self) -> None:
    import json
    from app.config import CONFIG_DIR, CONFIG_FILE

    token = self._token_entry.get().strip()
    CONFIG_DIR.mkdir(parents=True, exist_ok=True)
    data = {}
    if CONFIG_FILE.exists():
        try:
            data = json.loads(CONFIG_FILE.read_text())
        except (json.JSONDecodeError, OSError):
            pass
    data["groq_token"] = token
    CONFIG_FILE.write_text(json.dumps(data, indent=2))
    self._status_label.configure(text="Salvo!", text_color=ACCENT_TEAL)
    self.after(2000, lambda: self._status_label.configure(text=""))
```
