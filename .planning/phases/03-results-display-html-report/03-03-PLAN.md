**Complete new generator.py:**

```python
"""Geração do relatório HTML com Jinja2. Abre no browser local.
PDF export via WeasyPrint — deve rodar em Thread separada (nunca na thread Tkinter).
"""
from __future__ import annotations

import base64
import tempfile
import webbrowser
from datetime import datetime
from pathlib import Path
from threading import Thread
from typing import Callable, Optional

from jinja2 import Environment, FileSystemLoader

from app.core.models import ScanResult

TEMPLATE_DIR = Path(__file__).parent / "templates"
STATIC_DIR   = Path(__file__).parent / "static"
PROJECT_ROOT = Path(__file__).parent.parent.parent  # raksaas/

# Severity sort order for pre-sorting before template render
_SEVERITY_ORDER = {"critical": 0, "high": 1, "medium": 2, "low": 3, "info": 4}


def _load_chartjs() -> str:
    """Read bundled Chart.js from static dir. Returns empty string on failure."""
    path = STATIC_DIR / "chart.min.js"
    try:
        return path.read_text(encoding="utf-8")
    except OSError:
        return ""


def _load_logo_b64() -> str:
    """Encode Rakoon logo as base64 data URI. Returns empty string if file missing."""
    logo_path = PROJECT_ROOT / "rakoon_logo.png"
    try:
        data = logo_path.read_bytes()
        b64 = base64.b64encode(data).decode("ascii")
        return f"data:image/png;base64,{b64}"
    except OSError:
        return ""


def generate_report(result: ScanResult, open_browser: bool = True) -> Path:
    """Gera relatório HTML e (opcionalmente) abre no browser. Retorna path do arquivo.

    Injeta Chart.js inline (offline-safe, D-10), logo como base64 (D-09),
    findings pré-ordenados Critical-first (D-02), e timestamp de geração.
    """
    # Pre-sort findings Critical-first (template sort is secondary safety net only)
    sorted_findings = sorted(
        result.findings,
        key=lambda f: _SEVERITY_ORDER.get(f.severity.value, 99)
    )
    # Build a ScanResult-like namespace with sorted findings for template
    # We pass individual vars so template does not need to re-sort
    import dataclasses
    sorted_result = dataclasses.replace(result, findings=sorted_findings)

    env = Environment(
        loader=FileSystemLoader(str(TEMPLATE_DIR)),
        autoescape=False,  # HTML template, not user input
    )
    template = env.get_template("report.html")
    html = template.render(
        result=sorted_result,
        logo_b64=_load_logo_b64(),
        chartjs=_load_chartjs(),
        generated_at=datetime.now().strftime("%Y-%m-%d %H:%M"),
    )

    output = Path(tempfile.gettempdir()) / "secscan_report.html"
    output.write_text(html, encoding="utf-8")

    if open_browser:
        webbrowser.open(output.as_uri())

    return output


def export_pdf(
    result: ScanResult,
    output_path: Path,
    on_complete: Optional[Callable[[Path], None]] = None,
    on_error: Optional[Callable[[Exception], None]] = None,
) -> None:
    """Exporta o relatório como PDF via WeasyPrint.

    MUST be called from a non-Tkinter thread. Per D-15: WeasyPrint renders the
    same HTML as generate_report() → PDF. Per D-14: runs in daemon Thread.

    Args:
        result: ScanResult to render.
        output_path: Destination .pdf file path.
        on_complete: Called with output_path when done (use root.after to update GUI).
        on_error: Called with the exception on failure.
    """
    def _run() -> None:
        try:
            from weasyprint import HTML  # lazy import — weasyprint is heavy
            html_path = generate_report(result, open_browser=False)
            HTML(filename=str(html_path)).write_pdf(str(output_path))
            if on_complete:
                on_complete(output_path)
        except Exception as exc:
            if on_error:
                on_error(exc)

    t = Thread(target=_run, daemon=True)
    t.start()
```

**Key implementation notes:**
- `dataclasses.replace()` creates a shallow copy of ScanResult with sorted findings — this is safe because ScanResult is a dataclass.
- `autoescape=False` is intentional — report.html contains HTML markup, not untrusted user input.
- WeasyPrint is imported lazily inside `_run()` to avoid slowing down app startup.
- The `export_pdf()` function itself is non-blocking — it starts the thread and returns immediately.

```python
def _open_report(self) -> None:
    """Generate HTML report and open in browser. Runs in daemon thread (off Tkinter thread)."""
    if not self._result:
        return
    result = self._result

    def _run():
        try:
            generate_report(result, open_browser=True)
        except Exception as exc:
            # Schedule error dialog on main thread
            self.after(0, lambda: messagebox.showerror("Erro", f"Falha ao gerar relatório:\n{exc}"))

    threading.Thread(target=_run, daemon=True).start()

def _export_pdf(self) -> None:
    """Open save dialog and export PDF via WeasyPrint in a daemon thread. Per D-14, D-15, D-16."""
    if not self._result:
        return

    today = datetime.now().strftime("%Y%m%d")
    default_name = f"secscan_report_{today}.pdf"

    # asksaveasfilename must run on main thread (it is a Tkinter dialog)
    path_str = filedialog.asksaveasfilename(
        defaultextension=".pdf",
        filetypes=[("PDF files", "*.pdf"), ("All files", "*.*")],
        initialfile=default_name,
        title="Salvar relatório PDF",
    )
    if not path_str:
        return  # user cancelled

    from pathlib import Path
    output_path = Path(path_str)
    result = self._result

    def on_complete(p):
        self.after(0, lambda: messagebox.showinfo("PDF Exportado", f"Relatório salvo em:\n{p}"))

    def on_error(exc):
        self.after(0, lambda: messagebox.showerror("Erro PDF", f"Falha ao exportar PDF:\n{exc}"))

    export_pdf(result, output_path, on_complete=on_complete, on_error=on_error)
```
